cmake_minimum_required(VERSION 3.28)
set(CMAKE_SYSTEM_NAME WindowsStore)
set(CMAKE_SYSTEM_VERSION 10.0)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
project(ZeldaCrossoverUWP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WW_FEED "" CACHE PATH "Built Wind Waker UWP static libraries")
set(UWP_DEPS "" CACHE PATH "Pinned uwp-dep checkout")
set(PACKAGE_VERSION "0.1.0.1" CACHE STRING "Four-part app version")
if (NOT PACKAGE_VERSION MATCHES "^[0-9]+[.][0-9]+[.][0-9]+[.][0-9]+$")
    message(FATAL_ERROR "PACKAGE_VERSION must have four numeric parts; quote the -D argument in PowerShell")
endif ()
if (NOT EXISTS "${WW_FEED}/cos_uwp_game.lib")
    message(FATAL_ERROR "Build cos_uwp_game before configuring the package")
endif ()
if (NOT EXISTS "${UWP_DEPS}/x64/lib/SDL3.lib")
    message(FATAL_ERROR "UWP_DEPS must contain the pinned x64 libraries")
endif ()
add_executable(ZeldaCrossoverUWP WIN32 main.cpp)
set_property(TARGET ZeldaCrossoverUWP PROPERTY VS_WINDOWS_TARGET_PLATFORM_MIN_VERSION "10.0.19041.0")
target_include_directories(ZeldaCrossoverUWP PRIVATE "${UWP_DEPS}/x64/include")
target_compile_definitions(ZeldaCrossoverUWP PRIVATE COS_UWP=1 _UWP=1 NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ZeldaCrossoverUWP PRIVATE /EHsc)

file(GLOB_RECURSE _libs LIST_DIRECTORIES FALSE "${WW_FEED}/*.lib")
list(FILTER _libs EXCLUDE REGEX "[/\\\\]CMakeFiles[/\\\\]")
list(FILTER _libs EXCLUDE REGEX "[/\\\\](test|tests|examples)[/\\\\]")
list(FILTER _libs EXCLUDE REGEX "aurora_main[.]lib$")
list(REMOVE_ITEM _libs "${WW_FEED}/cos_uwp_game.lib")
set(_dlls SDL3 webgpu_dawn nod libuwp libpng16 z)
set(_imports)
foreach (_name IN LISTS _dlls)
    list(APPEND _imports "${UWP_DEPS}/x64/lib/${_name}.lib")
endforeach ()
target_link_libraries(ZeldaCrossoverUWP PRIVATE "${WW_FEED}/cos_uwp_game.lib"
    ${_libs} ${_imports} WindowsApp.lib OneCore.lib ntdll.lib DXGI.lib Ws2_32.lib)
target_link_options(ZeldaCrossoverUWP PRIVATE "/WHOLEARCHIVE:${WW_FEED}/cos_uwp_game.lib"
    /INCREMENTAL:NO /OPT:NOICF /NODEFAULTLIB:libcmt /NODEFAULTLIB:libcmtd)

configure_file(Package.appxmanifest.in "${CMAKE_CURRENT_BINARY_DIR}/Package.appxmanifest" @ONLY)
set(_manifest "${CMAKE_CURRENT_BINARY_DIR}/Package.appxmanifest")
set_source_files_properties("${_manifest}" PROPERTIES VS_DEPLOYMENT_CONTENT TRUE)
target_sources(ZeldaCrossoverUWP PRIVATE "${_manifest}")
foreach (_name IN LISTS _dlls ITEMS dxcompiler dxil)
    set(_dll "${UWP_DEPS}/x64/bin/${_name}.dll")
    if (NOT EXISTS "${_dll}")
        message(FATAL_ERROR "Missing UWP runtime: ${_dll}")
    endif ()
    set_source_files_properties("${_dll}" PROPERTIES VS_DEPLOYMENT_CONTENT TRUE
        VS_DEPLOYMENT_LOCATION "." VS_COPY_TO_OUT_DIR Always)
    target_sources(ZeldaCrossoverUWP PRIVATE "${_dll}")
endforeach ()
file(GLOB _art "${CMAKE_CURRENT_SOURCE_DIR}/Assets/*.png")
if (NOT _art)
    message(FATAL_ERROR "Run crossover/tools/package_assets.py before configuring")
endif ()
set_source_files_properties(${_art} PROPERTIES VS_DEPLOYMENT_CONTENT TRUE VS_DEPLOYMENT_LOCATION "Assets")
target_sources(ZeldaCrossoverUWP PRIVATE ${_art})
